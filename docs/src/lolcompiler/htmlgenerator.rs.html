2use crate::syntaxanalyzer::Node;
3use std::fs;
4use std::path::{Path, PathBuf};
5use std::process::Command;
6
7/// Converts the checked syntax tree into an HTML file.
8pub struct HtmlGenerator;
9
10impl HtmlGenerator {
11    /// Creates the HTML generator.
12    pub fn new() -> Self {
13        HtmlGenerator
14    }
15
16    // Replaces special characters so text does not get treated as HTML tags.
17    fn escape_html(text: &str) -> String {
18        text.replace('&', "&amp;")
19            .replace('<', "&lt;")
20            .replace('>', "&gt;")
21            .replace('"', "&quot;")
22    }
23
24    // Combines child nodes and adds spaces between pieces of visible text.
25    fn render_children(&self, node: &Node) -> String {
26        let mut result = String::new();
27        let mut previous_text = false;
28
29        for child in &node.children {
30            let text = self.render(child);
31
32            if text.is_empty() {
33                continue;
34            }
35
36            let current_text = matches!(
37                child.kind.as_str(),
38                "text" | "resolved_variable" | "bold" | "link"
39            );
40
41            if previous_text && current_text {
42                result.push(' ');
43            }
44
45            result.push_str(&text);
46            previous_text = current_text;
47        }
48
49        result
50    }
51
52    // Turns each syntax tree node into its matching HTML output.
53    fn render(&self, node: &Node) -> String {
54        match node.kind.as_str() {
55            "lolcode" => format!("<html>\n{}\n</html>\n", self.render_children(node)),
56            "head" => format!("<head>\n{}\n</head>\n", self.render_children(node)),
57            "title" => format!("<title>{}</title>\n", self.render_children(node)),
58            "paragraph" => format!("<p>{}</p>\n", self.render_children(node)),
59            "bold" => format!("<b>{}</b>", self.render_children(node)),
60            "list" => format!("<ul>\n{}\n</ul>\n", self.render_children(node)),
61            "inner_list" => format!("<li>{}</li>\n", self.render_children(node)),
62            "link" => {
63                let address = self.render_children(node);
64                format!("<a href=\"{}\">{}</a>", address, address)
65            }
66            "newline" => "<br>\n".to_string(),
67            "comment" => {
68                let content = node.children.iter()
69                    .filter(|child| child.kind == "comment_text")
70                    .filter_map(|child| child.token.as_ref())
71                    .cloned()
72                    .collect::<Vec<String>>()
73                    .join(" ");
74                format!("<!-- {} -->\n", content.replace("--", "- -"))
75            }
76            "variable_define" => String::new(),
77            "text" | "resolved_variable" => {
78                match &node.token {
79                    Some(token) => Self::escape_html(token),
80                    None => String::new(),
81                }
82            }
83            "variable_name" | "comment_text" => String::new(),
84            kind if kind.starts_with('#') => String::new(),
85            _ => self.render_children(node),
86        }
87    }
88
89    /// Writes the generated HTML next to the input file.
90    pub fn generate(&self, tree: &Node, filename: &str) -> Result<PathBuf, String> {
91        let input_path = Path::new(filename);
92        let output_path = input_path.with_extension("html");
93        let html = self.render(tree);
94
95        fs::write(&output_path, html)
96            .map_err(|e| format!("Error writing HTML: {}", e))?;
97
98        Ok(output_path)
99    }
100
101    /// Opens the generated HTML using the operating system's browser command.
102    pub fn open_browser(&self, path: &Path) -> Result<(), String> {
103        let full_path = path.canonicalize()
104            .map_err(|e| format!("Error finding HTML file: {}", e))?;
105
106        #[cfg(target_os = "windows")]
107        let result = Command::new("cmd")
108            .args(["/C", "start", ""])
109            .arg(&full_path)
110            .spawn();
111
112        #[cfg(target_os = "macos")]
113        let result = Command::new("open")
114            .arg(&full_path)
115            .spawn();
116
117        #[cfg(not(any(target_os = "windows", target_os = "macos")))]
118        let result = Command::new("xdg-open")
119            .arg(&full_path)
120            .spawn();
121
122        result
123            .map(|_| ())
124            .map_err(|e| format!("Error opening browser: {}", e))
125    }
126}
